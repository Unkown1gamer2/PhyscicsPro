    document.querySelectorAll(".status").forEach(select => {

      const saved = localStorage.getItem(`status-${id}`);
      if (saved) select.value = saved;

      const markInput = document.querySelector(`.mark[data-id='${id}']`);
      if (select.value === "Completed") markInput.classList.remove("hidden");
      const savedMark = localStorage.getItem(`mark-${id}`);
      if (savedMark) markInput.value = savedMark;

      select.addEventListener("change", () => {
        localStorage.setItem(`status-${id}`, select.value);
        if (select.value === "Completed") {
          markInput.classList.remove("hidden");
        } else {
          markInput.classList.add("hidden");
          localStorage.removeItem(`mark-${id}`);
