  <script>
    // Fill scroll pickers
    function fillPicker(el, max) {
      for (let i = 0; i <= max; i++) {
        let div = document.createElement("div");
        div.className = "picker-value";
        div.textContent = i.toString().padStart(2,"0");
        el.appendChild(div);
      }
    }
    fillPicker(document.getElementById("hours"), 23);
    fillPicker(document.getElementById("minutes"), 59);
    fillPicker(document.getElementById("seconds"), 59);

    const timerIcon = document.getElementById("timerIcon");
    const timerDropdown = document.getElementById("timerDropdown");
    const startBtn = document.getElementById("startTimer");
    const pauseBtn = document.getElementById("pauseTimer");
    const cancelBtn = document.getElementById("cancelTimer");

    let timerInterval;
    let remainingTime = 0;
    let paused = false;

    function getScrollValue(el) {
      const index = Math.round(el.scrollTop / 30);
      return index;
    }

    function formatTime(t) {
      const h = Math.floor(t / 3600).toString().padStart(2,"0");
      const m = Math.floor((t % 3600) / 60).toString().padStart(2,"0");
      const s = (t % 60).toString().padStart(2,"0");
      return `${h}:${m}:${s}`;
    }

    function updateNavbarTimerDisplay(show, text="") {
      if (show) {
        timerIcon.innerHTML = `<span class="timer-running">${text}</span>`;
      } else {
        timerIcon.textContent = "⏱️";
      }
    }

    timerIcon.addEventListener("click", () => {
      timerDropdown.style.display = (timerDropdown.style.display === "block") ? "none" : "block";
    });

    startBtn.addEventListener("click", () => {
      const h = getScrollValue(document.getElementById("hours"));
      const m = getScrollValue(document.getElementById("minutes"));
      const s = getScrollValue(document.getElementById("seconds"));
      remainingTime = h*3600 + m*60 + s;

      if (remainingTime <= 0) return;

      timerDropdown.style.display = "none";
      updateNavbarTimerDisplay(true, formatTime(remainingTime));

      timerInterval = setInterval(() => {
        if (!paused) {
          remainingTime--;
          if (remainingTime <= 0) {
            clearInterval(timerInterval);
            updateNavbarTimerDisplay(false);
            alert("⏱️ Time's up!");
          } else {
            updateNavbarTimerDisplay(true, formatTime(remainingTime));
          }
        }
      }, 1000);
    });

    pauseBtn.addEventListener("click", () => {
      paused = !paused;
      pauseBtn.textContent = paused ? "Resume" : "Pause";
    });

    cancelBtn.addEventListener("click", () => {
      clearInterval(timerInterval);
      updateNavbarTimerDisplay(false);
      timerDropdown.style.display = "none";
      paused = false;
      pauseBtn.textContent = "Pause";
    });

    // Scroll to footer
    document.getElementById("scrollDown").addEventListener("click", () => {
      window.scrollTo({ top: document.body.scrollHeight, behavior: "smooth" });
    });
  </script>
