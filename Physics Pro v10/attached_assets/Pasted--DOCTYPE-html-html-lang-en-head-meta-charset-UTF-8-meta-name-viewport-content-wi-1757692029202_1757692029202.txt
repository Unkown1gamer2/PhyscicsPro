<script>
const easing = {
  easeOutCubic: function(pos) {
    return (Math.pow((pos-1), 3) +1);
  }
};

class IosSelector {
  constructor(options) {
    let defaults = {
      el: '', 
      type: 'infinite', 
      count: 20, 
      source: [], 
      value: null,
      onChange: null
    };
    this.options = Object.assign({}, defaults, options);
    this.options.count =  this.options.count - this.options.count % 4;
    Object.assign(this, this.options);

    this.halfCount = this.options.count / 2;
    this.quarterCount = this.options.count / 4;
    this.itemHeight = 50; 
    this.itemAngle = 360 / this.options.count; 
    this.radius = this.itemHeight / Math.tan(this.itemAngle * Math.PI / 180); 
    this.scroll = 0;
    this.moving = false;

    this.elems = { el: document.querySelector(this.options.el) };
    this._create(this.options.source);
  }

  _create(source) {
    if (!source.length) return;
    this.source = source;
    let circleListHTML = '';
    for (let i = 0; i < source.length; i++) {
      circleListHTML += `<li class="select-option" data-index="${i}" style="top: ${-this.itemHeight/2}px;height:${this.itemHeight}px;line-height:${this.itemHeight}px;transform: rotateX(${-this.itemAngle * i}deg) translateZ(${this.radius}px);">${source[i]}</li>`;
    }
    this.elems.el.innerHTML = `<div class="select-wrap"><ul class="select-options">${circleListHTML}</ul><div class="highlight"><ul class="highlight-list">${circleListHTML}</ul></div></div>`;
    this.elems.circleList = this.elems.el.querySelector('.select-options');
    this.elems.circleItems = this.elems.el.querySelectorAll('.select-option');
    this.elems.highlightList = this.elems.el.querySelectorAll('.highlight-item');
  }

  select(value) {
    let index = this.source.indexOf(value);
    if (index === -1) index = 0;
    this.scroll = index;
    this._update();
    if (this.onChange) this.onChange(this.source[this.scroll]);
  }

  _update() {
    this.elems.circleList.style.transform = `translateZ(${-this.radius}px) rotateX(${this.itemAngle*this.scroll}deg)`;
  }

  getValue() {
    return this.source[this.scroll];
  }
}

// Build sources
let hoursSource = [];
for (let i=0;i<=5;i++) hoursSource.push(i);
let minutesSecondsSource = [];
for (let i=0;i<=59;i++) minutesSecondsSource.push(i);

// Initialize selectors
let hourSelector = new IosSelector({el:'#hour-selector', source: hoursSource});
let minuteSelector = new IosSelector({el:'#minute-selector', source: minutesSecondsSource});
let secondSelector = new IosSelector({el:'#second-selector', source: minutesSecondsSource});

// Timer logic
let timerInterval = null;
function startTimer() {
  let totalSeconds = hourSelector.getValue()*3600 + minuteSelector.getValue()*60 + secondSelector.getValue();
  if (totalSeconds <= 0) return;
  const display = document.getElementById('timerDisplay');

  timerInterval = setInterval(() => {
    if (totalSeconds <= 0) {
      clearInterval(timerInterval);
      display.textContent = "00:00:00";
      return;
    }
    totalSeconds--;
    let h = Math.floor(totalSeconds/3600);
    let m = Math.floor((totalSeconds%3600)/60);
    let s = totalSeconds%60;
    display.textContent = `${String(h).padStart(2,'0')}:${String(m).padStart(2,'0')}:${String(s).padStart(2,'0')}`;
  }, 1000);
}

document.getElementById('startTimerBtn').addEventListener('click', startTimer);
